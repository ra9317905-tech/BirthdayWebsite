  <script>
    // Initialize Lucide Icons
    lucide.createIcons();

    // ==========================================
    // 1. CONFIGURATION & CUSTOMIZATION
    // ==========================================
    const CONFIG = {
      recipientName: "Maya",
      senderName: "Alex",
      notes: [
        "The world is simply softer, kinder, and infinitely more joyful with your laughter echoing in it.",
        "Your empathy is a quiet superpower. Never forget how profoundly you touch lives.",
        "May this year give you courage to chase that dream you only whisper about when it's dark.",
        "Thank you for being you — completely unfiltered, wonderfully spontaneous, and deeply genuine.",
        "Every chapter ahead of you is going to be richer and bolder than the last."
      ]
    };

    // Update names dynamically
    document.getElementById('recipient-name').textContent = CONFIG.recipientName;

    // ==========================================
    // 2. WEB AUDIO API SYNTHESIZER (No 404s!)
    // ==========================================
    let audioCtx = null;
    let isSoundOn = false;
    let ambientOscillator = null;

    function initAudio() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
    }

    // Play pleasant synthetic chime
    function playChime(freq = 523.25, type = 'sine', duration = 0.8) {
      if (!isSoundOn) return;
      initAudio();
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();

        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);

        osc.connect(gain);
        gain.connect(audioCtx.destination);

        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {
        console.warn("Audio play prevented:", e);
      }
    }

    function playChord() {
      if (!isSoundOn) return;
      [440, 554.37, 659.25, 880].forEach((freq, idx) => {
        setTimeout(() => playChime(freq, 'sine', 1.2), idx * 120);
      });
    }

    // Sound toggle button
    const audioBtn = document.getElementById('audio-toggle');
    const audioLabel = document.getElementById('audio-label');

    audioBtn.addEventListener('click', () => {
      initAudio();
      isSoundOn = !isSoundOn;
      if (isSoundOn) {
        audioLabel.textContent = "Sound On";
        audioBtn.classList.add('playing');
        playChord();
      } else {
        audioLabel.textContent = "Muted";
        audioBtn.classList.remove('playing');
      }
    });

    // ==========================================
    // 3. UNLOCK ENTRY GATE
    // ==========================================
    const entryGate = document.getElementById('entry-gate');

    const unlockProgress = document.getElementById('unlock-progress');

    // ==========================================
    // 7. AMBIENT BACKGROUND & LANTERN ENGINE
    // ==========================================
    const canvas = document.getElementById('ambient-canvas');
    const ctx = canvas.getContext('2d');

    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    window.addEventListener('resize', () => {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    });

    // Stars
    const stars = Array.from({ length: 140 }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      radius: Math.random() * 1.3,
      alpha: Math.random(),
      twinkleSpeed: 0.005 + Math.random() * 0.015
    }));

    // Floating Sky Lanterns / Wishes
    const lanterns = [];

    function addLantern(text = "") {
      lanterns.push({
        x: width * 0.3 + Math.random() * (width * 0.4),
        y: height + 20,
        targetX: Math.random() * width,
        vy: 1.2 + Math.random() * 0.8,
        vx: (Math.random() - 0.5) * 0.5,
        radius: 4 + Math.random() * 3,
        alpha: 1,
        text: text
      });
    }

    // Animation Loop
    function animate() {
      ctx.clearRect(0, 0, width, height);

      // Render stars
      stars.forEach(star => {
        star.alpha += star.twinkleSpeed;
        if (star.alpha > 1 || star.alpha < 0.2) star.twinkleSpeed = -star.twinkleSpeed;
        
        ctx.beginPath();
        ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(255, 255, 255, ${Math.max(0, star.alpha)})`;
        ctx.fill();
      });

      // Render sky lanterns
      for (let i = lanterns.length - 1; i >= 0; i--) {
        const l = lanterns[i];
        l.y -= l.vy;
        l.x += l.vx;
        
        // Warm Lantern Glow
        const radGrad = ctx.createRadialGradient(l.x, l.y, 0, l.x, l.y, l.radius * 4);
        radGrad.addColorStop(0, 'rgba(254, 215, 170, 0.9)');
        radGrad.addColorStop(0.4, 'rgba(249, 115, 22, 0.4)');
        radGrad.addColorStop(1, 'transparent');

        ctx.fillStyle = radGrad;
        ctx.beginPath();
        ctx.arc(l.x, l.y, l.radius * 4, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.arc(l.x, l.y, l.radius * 0.8, 0, Math.PI * 2);
        ctx.fill();

        // If lantern carries custom text
        if (l.text) {
          ctx.font = '11px "Plus Jakarta Sans"';
          ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
          ctx.fillText(l.text, l.x + 12, l.y + 4);
        }

        // Remove lanterns that drifted off top
        if (l.y < -50) lanterns.splice(i, 1);
      }

      requestAnimationFrame(animate);
    }
    animate();

    // ==========================================

    const wishForm = document.getElementById('wish-form');
    const wishInput = document.getElementById('wish-input');

      e.preventDefault();
      const val = wishInput.value.trim();
      if (!val) return;

      addLantern(val);
      playChime(784, 'sine', 0.7);

      totalWishes++;
      wishCounterText.textContent = `${totalWishes} Wish${totalWishes > 1 ? 'es' : ''} floating in our sky`;

      confetti({
        particleCount: 40,
        spread: 60,
        origin: { y: 0.85 },
        colors: ['#38bdf8', '#818cf8', '#f472b6']
      });

      wishInput.value = '';
    });
